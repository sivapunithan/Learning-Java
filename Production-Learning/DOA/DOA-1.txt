public List<LineItemDOAResponseDTO> getLineItemWiseApprovers(int treqId, List<Integer> excludedEmpIds) {

		TRequestEntity request = treq.findByReqId(treqId).orElseThrow(() -> new RuntimeException("Request not found"));

		List<TRequestAssetEntity> assets = treqAsset.findAllByReqId(treqId);

		if (assets.isEmpty()) {
			throw new RuntimeException("No assets found");
		}

		List<LineItemDOAResponseDTO> response = new ArrayList<>();

		for (TRequestAssetEntity asset : assets) {

			List<EmployeeIpaApprovalDTO> approvers = getApproversForAsset(request, asset, excludedEmpIds);

			LineItemDOAResponseDTO dto = new LineItemDOAResponseDTO();
			dto.setReqAssetId(asset.getReqAssetId());
			dto.setReqAssetName(asset.getProductName());
			dto.setCcId(asset.getCcId());
			dto.setCategoryId(asset.getCategoryId());
			dto.setAmount(asset.getTotalPrice());
			dto.setApprovers(approvers);

			response.add(dto);
		}
		return response;
	}

    public List<EmployeeIpaApprovalDTO> getApproversForAsset(
            TRequestEntity request,
            TRequestAssetEntity asset,
            List<Integer> excludedEmpIds) {

        BigDecimal amount = asset.getTotalPrice();

        List<Integer> officeTypeId =
                Arrays.stream(request.getOfficeTypeIds().split(","))
                        .map(String::trim)
                        .map(Integer::valueOf)
                        .distinct()
                        .collect(Collectors.toList());

        Integer entityId = request.getCompId();

        if (excludedEmpIds == null || excludedEmpIds.isEmpty()) {
            excludedEmpIds = Collections.singletonList(-1);
        }

        EmployeeEntity requester = employeeMasterRepository
                .findById(request.getPrRaisedby())
                .orElseThrow(() -> new RuntimeException("Requester not found"));

        BigDecimal requesterLimit =
                dOAMatrixRepo.findRequesterLimitForLineItem(
                        requester.getEmpId(),
                        officeTypeId,
                        entityId,
                        asset.getCcId(),
                        asset.getCategoryId()
                );

        boolean requesterCanApprove =
                requesterLimit != null && amount.compareTo(requesterLimit) <= 0;
        // GRADE FALL-BACK
        if (requesterCanApprove) {
            int nextGrade = gradeMasterRepo.findGradeHierarchy(requester.getGradeId()) + 1;
            Integer maxGrade = gradeMasterRepo.findMaxGradeHierarchy();

            while (nextGrade <= maxGrade) {
                List<EmployeeIpaApprovalDTO> approvers =
                        employeeMasterRepository.findApproversByExactGrade(
                                nextGrade, requester.getEmpId()
                        );

                if (!approvers.isEmpty()) {
                    return approvers;
                }
                nextGrade++;
            }
        }

        List<EmployeeIpaApprovalDTO> approvers =
                dOAMatrixRepo.findApproversForSingleLineItem(
                        officeTypeId,
                        entityId,
                        asset.getCcId(),
                        asset.getCategoryId(),
                        amount,
                        excludedEmpIds
                );

        if (approvers.isEmpty()) {
            throw new RuntimeException(
                    "No DOA approver configured for asset " + asset.getReqAssetId()
            );
        }

        return approvers;
    }
